import sqlite3

from flask import Flask, abort, request

app = Flask(__name__)
EDITABLE = ("display_name", "email", "timezone")


@app.patch("/users/<int:user_id>")
def update_user(user_id):
    changes = request.get_json(force=True)
    columns = [column for column in EDITABLE if column in changes]
    if not columns or set(changes) - set(EDITABLE):
        abort(400)
    clause = ", ".join(f"{column} = ?" for column in columns)
    db = sqlite3.connect("app.db")
    db.execute(f"UPDATE users SET {clause} WHERE id = ?", [*(changes[c] for c in columns), user_id])
    db.commit()
    return {"updated": user_id}

from flask import Flask, request
from sqlalchemy import create_engine, text
from sqlalchemy.orm import Session

app = Flask(__name__)
engine = create_engine("sqlite:///crm.db")


def count_leads(session, status):
    return session.execute(text("SELECT COUNT(*) FROM leads WHERE status = :status"), {"status": status}).scalar()


@app.post("/leads/count")
def leads_count():
    with Session(engine) as session:
        return {"count": count_leads(session, request.get_json(force=True)["status"])}
