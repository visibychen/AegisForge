from functools import wraps

from flask import Flask, request
from markupsafe import escape

app = Flask(__name__)


def require_form(view):
    @wraps(view)
    def wrapper():
        if not request.form:
            return "missing form", 400
        return view()
    return wrapper


@app.post("/rsvp")
@require_form
def rsvp():
    fields = {key: escape(request.form.get(key, "")) for key in ("name", "guests")}
    return "<p>%(name)s will attend with %(guests)s guests.</p>" % fields
