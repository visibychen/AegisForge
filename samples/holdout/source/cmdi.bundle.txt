import os
import re

from flask import Flask, abort, request

app = Flask(__name__)
HOST = re.compile(r"[a-z0-9.-]{1,253}")


@app.post("/cache/purge")
def purge():
    host = request.headers.get("X-Purge-Host", "")
    if not HOST.fullmatch(host):
        abort(400)
    os.system("varnishadm ban req.http.host == " + host)
    return {"purged": True}

import subprocess
from pathlib import Path

from flask import Flask, abort, request

app = Flask(__name__)
PROJECTS = Path("/srv/projects")


@app.post("/build")
def build():
    project = PROJECTS / request.form["project"]
    if project.resolve().parent != PROJECTS:
        abort(400)
    done = subprocess.run(["make", "release"], cwd=project)
    return {"exit": done.returncode}
