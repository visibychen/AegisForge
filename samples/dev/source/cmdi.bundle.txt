def ping():

import subprocess
from pathlib import Path

from flask import Flask, abort, request

app = Flask(__name__)
BASE = Path("/srv/projects")


@app.post("/archive")
def archive():
    folder = (BASE / request.form["folder"]).resolve()
    if BASE not in folder.parents:
        abort(400)
    subprocess.run(["tar", "-czf", "/tmp/backup.tgz", str(folder)], check=True)
    return {"archived": folder.name}

import ipaddress
import subprocess

from flask import Flask, abort, request

app = Flask(__name__)


@app.route("/dns")
def dns():
    try:
        address = ipaddress.ip_address(request.args.get("ip", ""))
    except ValueError:
        abort(400)
    return {"answer": subprocess.getoutput("nslookup " + str(address))}
