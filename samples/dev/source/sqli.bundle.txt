import sqlite3

from flask import Flask, request

app = Flask(__name__)


@app.route("/books")
def books():
    term = request.args.get("term", "")
    db = sqlite3.connect("library.db")
    pattern = "%" + term.replace("%", r"\%").replace("_", r"\_") + "%"
    sql = "SELECT title FROM books WHERE title LIKE ? ESCAPE '\\'"
    return {"titles": [r[0] for r in db.execute(sql, (pattern,))]}
